    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Clock functionality
            function updateClock() {
                const now = new Date();
                const timeEl = document.getElementById('currentTime');
                const dateEl = document.getElementById('currentDate');
                
                // Format time (HH:MM:SS)
                const hours = now.getHours().toString().padStart(2, '0');
                const minutes = now.getMinutes().toString().padStart(2, '0');
                const seconds = now.getSeconds().toString().padStart(2, '0');
                timeEl.textContent = `${hours}:${minutes}:${seconds}`;
                
                // Format date (Weekday, Month Day)
                const options = { weekday: 'long', month: 'long', day: 'numeric' };
                dateEl.textContent = now.toLocaleDateString('en-US', options);
            }
            
            updateClock();
            setInterval(updateClock, 1000);
            
            // Timer functionality
            let timerInterval;
            let seconds = 0;
            let isRunning = false;
            let isPaused = false;
            
            function formatTime(totalSeconds) {
                const hours = Math.floor(totalSeconds / 3600).toString().padStart(2, '0');
                const minutes = Math.floor((totalSeconds % 3600) / 60).toString().padStart(2, '0');
                const secs = (totalSeconds % 60).toString().padStart(2, '0');
                return { hours, minutes, seconds: secs };
            }
            
            function updateTimerDisplay() {
                const { hours, minutes, seconds: secs } = formatTime(seconds);
                const timerDisplay = document.getElementById('timerDisplay');
                timerDisplay.querySelector('.hours').textContent = hours;
                timerDisplay.querySelector('.minutes').textContent = minutes;
                timerDisplay.querySelector('.seconds').textContent = secs;
            }
            
            function startTimer() {
                if (isRunning) return;
                
                isRunning = true;
                isPaused = false;
                document.getElementById('timerState').textContent = 'Tracking Time';
                document.getElementById('startTimerBtn').classList.add('hidden');
                document.getElementById('timerControls').classList.remove('hidden');
                
                // Start from 0 if not paused
                if (!isPaused) {
                    seconds = 0;
                    updateTimerDisplay();
                }
                
                timerInterval = setInterval(() => {
                    seconds++;
                    updateTimerDisplay();
                }, 1000);
            }
            
            function pauseTimer() {
                if (!isRunning) return;
                
                clearInterval(timerInterval);
                isRunning = false;
                isPaused = true;
                document.getElementById('timerState').textContent = 'Paused';
                document.getElementById('pauseTimerBtn').textContent = 'Resume';
            }
            
            function resumeTimer() {
                if (isRunning) return;
                
                isRunning = true;
                isPaused = false;
                document.getElementById('timerState').textContent = 'Tracking Time';
                document.getElementById('pauseTimerBtn').textContent = 'Pause';
                
                timerInterval = setInterval(() => {
                    seconds++;
                    updateTimerDisplay();
                }, 1000);
            }
            
            function stopTimer() {
                clearInterval(timerInterval);
                isRunning = false;
                isPaused = false;
                
                // Save session
                const project = document.getElementById('projectSelect').value;
                const task = document.getElementById('taskInput').value || 'No description';
                const { hours, minutes, seconds: secs } = formatTime(seconds);
                const duration = `${hours}:${minutes}:${secs}`;
                
                addSessionToHistory(project, task, duration);
                
                // Reset UI
                document.getElementById('timerState').textContent = 'Current Session';
                document.getElementById('startTimerBtn').classList.remove('hidden');
                document.getElementById('timerControls').classList.add('hidden');
                seconds = 0;
                updateTimerDisplay();
                document.getElementById('taskInput').value = '';
            }
            
            function addSessionToHistory(project, task, duration) {
                const sessionList = document.getElementById('sessionList');
                const colors = ['#FF6B6B', '#48BB78', '#4299E1', '#9F7AEA', '#ED8936'];
                const randomColor = colors[Math.floor(Math.random() * colors.length)];
                
                const sessionItem = document.createElement('div');
                sessionItem.className = 'session-item';
                sessionItem.innerHTML = `
                    <div class="session-info">
                        <span class="project-tag" style="background: ${randomColor};">${project}</span>
                        <p>${task}</p>
                    </div>
                    <div class="session-time">
                        ${duration}
                    </div>
                    <button class="icon-btn">
                        <svg viewBox="0 0 24 24" width="16" height="16" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                            <circle cx="12" cy="12" r="1"></circle>
                            <circle cx="12" cy="5" r="1"></circle>
                            <circle cx="12" cy="19" r="1"></circle>
                        </svg>
                    </button>
                `;
                
                sessionList.insertBefore(sessionItem, sessionList.firstChild);
                
                // Limit to 10 sessions
                if (sessionList.children.length > 10) {
                    sessionList.removeChild(sessionList.lastChild);
                }
            }
            
            // Timer event listeners
            document.getElementById('startTimerBtn').addEventListener('click', startTimer);
            document.getElementById('pauseTimerBtn').addEventListener('click', function() {
                if (isPaused) {
                    resumeTimer();
                } else {
                    pauseTimer();
                }
            });
            document.getElementById('stopTimerBtn').addEventListener('click', stopTimer);
            document.getElementById('addMinuteBtn').addEventListener('click', function() {
                seconds += 60;
                updateTimerDisplay();
            });
            document.getElementById('subtractMinuteBtn').addEventListener('click', function() {
                if (seconds >= 60) {
                    seconds -= 60;
                    updateTimerDisplay();
                }
            });
            document.getElementById('flagTimeBtn').addEventListener('click', function() {
                const project = document.getElementById('projectSelect').value;
                const task = document.getElementById('taskInput').value || 'Flagged time';
                const { hours, minutes, seconds: secs } = formatTime(seconds);
                const duration = `${hours}:${minutes}:${secs}`;
                
                addSessionToHistory(project, task, duration);
                
                // Reset timer but keep it running
                seconds = 0;
                updateTimerDisplay();
                document.getElementById('taskInput').value = '';
                
                // Visual feedback
                const timerDisplay = document.getElementById('timerDisplay');
                timerDisplay.classList.add('pulse');
                setTimeout(() => {
                    timerDisplay.classList.remove('pulse');
                }, 2000);
            });
            
            // Music player functionality
            let isPlaying = false;
            const playlists = {
                'focus-flow': {
                    name: 'Focus Flow',
                    playlist: 'Deep Work Playlist',
                    cover: 'https://source.unsplash.com/random/200x200/?album,electronic'
                },
                'productive-coding': {
                    name: 'Algorithm',
                    playlist: 'Productive Coding',
                    cover: 'https://source.unsplash.com/random/200x200/?album,techno'
                },
                'creative-thinking': {
                    name: 'Brainstorm',
                    playlist: 'Creative Thinking',
                    cover: 'https://source.unsplash.com/random/200x200/?album,ambient'
                }
            };
            let currentPlaylist = 'focus-flow';
            
            function updatePlayerDisplay() {
                const playlist = playlists[currentPlaylist];
                document.getElementById('currentTrack').textContent = playlist.name;
                document.getElementById('currentPlaylist').textContent = playlist.playlist;
                document.querySelector('.album-art img').src = playlist.cover;
            }
            
            function togglePlayPause() {
                isPlaying = !isPlaying;
                const playPauseBtn = document.getElementById('playPauseBtn');
                
                if (isPlaying) {
                    playPauseBtn.innerHTML = `
                        <svg viewBox="0 0 24 24" width="24" height="24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                            <rect x="6" y="4" width="4" height="16"></rect>
                            <rect x="14" y="4" width="4" height="16"></rect>
                        </svg>
                    `;
                    // Simulate progress
                    simulateProgress();
                } else {
                    playPauseBtn.innerHTML = `
                        <svg viewBox="0 0 24 24" width="24" height="24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                            <polygon points="5 3 19 12 5 21 5 3"></polygon>
                        </svg>
                    `;
                    clearTimeout(progressTimeout);
                }
            }
            
            let progressTimeout;
            let progress = 0;
            
            function simulateProgress() {
                if (!isPlaying) return;
                
                progress += 1;
                if (progress > 100) {
                    progress = 0;
                    // Simulate track change
                    if (Math.random() > 0.7) {
                        changeTrack();
                    }
                }
                
                document.getElementById('progress').style.width = `${progress}%`;
                
                // Update time display
                const totalSeconds = 204; // 3:24
                const elapsedSeconds = Math.floor(totalSeconds * (progress / 100));
                const minutes = Math.floor(elapsedSeconds / 60);
                const secs = elapsedSeconds % 60;
                document.getElementById('currentTimeElapsed').textContent = `${minutes}:${secs.toString().padStart(2, '0')}`;
                
                progressTimeout = setTimeout(simulateProgress, 100);
            }
            
            function changeTrack() {
                const playlistKeys = Object.keys(playlists);
                currentPlaylist = playlistKeys[Math.floor(Math.random() * playlistKeys.length)];
                updatePlayerDisplay();
                progress = 0;
            }
            
            // Music player event listeners
            document.getElementById('playPauseBtn').addEventListener('click', togglePlayPause);
            document.getElementById('prevTrackBtn').addEventListener('click', function() {
                changeTrack();
                if (isPlaying) {
                    progress = 0;
                    document.getElementById('progress').style.width = '0%';
                }
            });
            document.getElementById('nextTrackBtn').addEventListener('click', function() {
                changeTrack();
                if (isPlaying) {
                    progress = 0;
                    document.getElementById('progress').style.width = '0%';
                }
            });

                // In a real app, this would control actual audio volume
                console.log('Volume set to:', e.target.value);
            });
            document.querySelectorAll('.playlist-item').forEach(item => {
                item.addEventListener('click', function() {
                    currentPlaylist = this.dataset.playlist;
                    updatePlayerDisplay();
                    if (!isPlaying) {
                        togglePlayPause();
                    }
                    progress = 0;
                    document.getElementById('progress').style.width = '0%';
                });
            });
            
            // Progress bar seeking
            document.getElementById('progressBar').addEventListener('click', function(e) {
                const rect = this.getBoundingClientRect();
                const pos = (e.clientX - rect.left) / rect.width;
                progress = Math.min(100, Math.max(0, pos * 100));
                document.getElementById('progress').style.width = `${progress}%`;
                
                if (isPlaying) {
                    clearTimeout(progressTimeout);
                    simulateProgress();
                }
            });
            
            // Notes functionality
            let notes = JSON.parse(localStorage.getItem('tickd-notes')) || [];
            
            function renderNotes() {
                const notesContainer = document.getElementById('savedNotes');
                notesContainer.innerHTML = '';
                
                notes.forEach(note => {
                    const noteEl = document.createElement('div');
                    noteEl.className = 'note-item';
                    noteEl.innerHTML = `
                        <p class="note-date">${note.date}</p>
                        <p class="note-content">${note.content}</p>
                    `;
                    notesContainer.appendChild(noteEl);
                });
            }
            
            function saveNote() {
                const content = document.getElementById('noteInput').value.trim();
                if (!content) return;
                
                const now = new Date();
                const options = { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' };
                const date = now.toLocaleDateString('en-US', options);
                
                notes.unshift({ date, content });
                localStorage.setItem('tickd-notes', JSON.stringify(notes));
                
                document.getElementById('noteInput').value = '';
                renderNotes();
            }
            
            // Notes event listeners
            document.getElementById('saveNoteBtn').addEventListener('click', saveNote);
            document.getElementById('noteInput').addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && e.ctrlKey) {
                    saveNote();
                }
            });
            
            // Tab switching
            document.querySelectorAll('.tab-btn').forEach(tab => {
                tab.addEventListener('click', function() {
                    // Update active tab
                    document.querySelectorAll('.tab-btn').forEach(t => t.classList.remove('active'));
                    this.classList.add('active');
                    
                    // Show corresponding section
                    const tabId = this.dataset.tab;
                    document.querySelectorAll('.panel-section').forEach(section => {
                        section.classList.remove('active');
                    });
                    document.getElementById(`${tabId}Section`).classList.add('active');
                });
            });
            
            // Panel toggle
            document.getElementById('togglePanelBtn').addEventListener('click', function() {
                const panel = document.querySelector('.creativity-panel');
                const isHidden = panel.style.transform === 'translateX(100%)';
                
                if (isHidden) {
                    panel.style.transform = 'translateX(0)';
                    this.innerHTML = `
                        <svg viewBox="0 0 24 24" width="20" height="20" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                            <polyline points="15 18 9 12 15 6"></polyline>
                        </svg>
                    `;
                } else {
                    panel.style.transform = 'translateX(100%)';
                    this.innerHTML = `
                        <svg viewBox="0 0 24 24" width="20" height="20" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                            <polyline points="9 18 15 12 9 6"></polyline>
                        </svg>
                    `;
                }
            });

            updatePlayerDisplay();
            renderNotes();
