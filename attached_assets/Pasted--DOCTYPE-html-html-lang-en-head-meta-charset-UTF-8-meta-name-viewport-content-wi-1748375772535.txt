    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const countdownDate = new Date('June 30, 2025 12:00:00').getTime();
            let currentDays = 0, currentHours = 0, currentMinutes = 0, currentSeconds = 0;
            let targetDays = 0, targetHours = 0, targetMinutes = 0, targetSeconds = 0;
            let animationFrameId = null;
            
            const countdownInterval = setInterval(function() {
                const now = new Date().getTime();
                const distance = countdownDate - now;
                
                targetDays = Math.floor(distance / (1000 * 60 * 60 * 24));
                targetHours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                targetMinutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
                targetSeconds = Math.floor((distance % (1000 * 60)) / 1000);
                
                if (!animationFrameId) {
                    animateCountdown();
                }
                
                if (distance < 0) {
                    clearInterval(countdownInterval);
                    document.getElementById('days').innerHTML = '0';
                    document.getElementById('hours').innerHTML = '0';
                    document.getElementById('minutes').innerHTML = '0';
                    document.getElementById('seconds').innerHTML = '0';
                    cancelAnimationFrame(animationFrameId);
                    createConfetti(); 
                }
            }, 1000);
            
            function animateCountdown() {
                const diffDays = targetDays - currentDays;
                const diffHours = targetHours - currentHours;
                const diffMinutes = targetMinutes - currentMinutes;
                const diffSeconds = targetSeconds - currentSeconds;
                
                currentDays += diffDays * 0.1;
                currentHours += diffHours * 0.1;
                currentMinutes += diffMinutes * 0.1;
                currentSeconds += diffSeconds * 0.1;
                
                document.getElementById('days').innerHTML = Math.floor(currentDays);
                document.getElementById('hours').innerHTML = Math.floor(currentHours);
                document.getElementById('minutes').innerHTML = Math.floor(currentMinutes);
                document.getElementById('seconds').innerHTML = Math.floor(currentSeconds);
                
                if (Math.abs(diffDays) > 0.1 || Math.abs(diffHours) > 0.1 || 
                    Math.abs(diffMinutes) > 0.1 || Math.abs(diffSeconds) > 0.1) {
                    animationFrameId = requestAnimationFrame(animateCountdown);
                } else {
                    currentDays = targetDays; currentHours = targetHours;
                    currentMinutes = targetMinutes; currentSeconds = targetSeconds;
                    document.getElementById('days').innerHTML = Math.floor(currentDays);
                    document.getElementById('hours').innerHTML = Math.floor(currentHours);
                    document.getElementById('minutes').innerHTML = Math.floor(currentMinutes);
                    document.getElementById('seconds').innerHTML = Math.floor(currentSeconds);
                    animationFrameId = null;
                }
            }
            
            function createConfetti() {
                const colors = ['#007AFF', '#34C759', '#5856D6', '#FF2D55', '#FF9500'];
                for (let i = 0; i < 100; i++) {
                    const confetti = document.createElement('div');
                    confetti.className = 'confetti';
                    confetti.style.left = Math.random() * 100 + 'vw';
                    confetti.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
                    confetti.style.width = Math.random() * 10 + 5 + 'px';
                    confetti.style.height = Math.random() * 10 + 5 + 'px';
                    confetti.style.animationDuration = Math.random() * 3 + 2 + 's';
                    confetti.style.animationDelay = Math.random() * 2 + 's';
                    document.body.appendChild(confetti);
                    setTimeout(() => { confetti.remove(); }, 5000);
                }
            }
        });
